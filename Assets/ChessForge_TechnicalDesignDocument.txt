ZeplinPrototipe - Complete Project Guide
A comprehensive developer's guide to understanding the ZeplinPrototipe Unity card battle game
🎮 Project Overview
ZeplinPrototipe is a turn-based card battle game featuring zeppelin aerial combat with resource management mechanics. Players command a crew aboard a zeppelin, manage four types of resources (Scrap, Water, Energy, Plant), and engage in grid-based tactical battles against enemy forces through strategic card placement and action execution.
Technical Foundation:
Unity Version: 6000.0 (Unity 6)
Render Pipeline: Universal Render Pipeline (URP) 17.0.3
Architecture: Manager-based singleton pattern with event-driven communication
Save System: Easy Save 3 integration for persistence
UI Framework: UGUI with DOTween animations

public void AddResource(ResourceType resourceType, int value = 1)
{
Resources[resourceType] += value;

    if (!resourceDisplays.ContainsKey(resourceType))
    {
        // Spawn new UI element for first-time resources
        GameObject go = Instantiate(ImagePrefab, ImageParent);
        Image img = go.GetComponent<Image>();
        img.sprite = _resourceIcons.Icons[resourceType];

        // Configure text display
        TextMeshProUGUI countText = go.GetComponentInChildren<TextMeshProUGUI>();
        resourceDisplays[resourceType] = new ResourceDisplay
        {
            image = img,
            countText = countText
        };
    }

    // Update displayed count
    resourceDisplays[resourceType].countText.text = Resources[resourceType].ToString();

}

public bool CompareResources()
{
if (CurrentCoolDown != 0) return false;

    if (effectHandler.Effects.Any(x => x is FreeResourceEffect))
        return true;

    for (int i = 1; i < (int)ResourceType.ResourceCount; i++)
    {
        if (RequiredResources[(ResourceType)i] > unitScript.Resources[(ResourceType)i])
            return false;
    }
    return true;

}
